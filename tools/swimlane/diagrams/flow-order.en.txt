[role_cust: Send the order] <block_neutral>
label: Order;
desc: The order arrives by email, fax or PDF. A fax is scanned on the copier into a file.;
if (Entry) is (File) than #blue
  if (Route) is (Email) than #blue
    [role_system: Receive the email] <block_system>
    label: Email;
    desc: Attachments (PDF, images) of mail to order-intake@ go into the intake folder. One attachment = one order acceptance. Production only.;
  elseif (Folder) than #blue
    [role_admin: Drop into folder] <block_apply>
    label: Drop;
    props: APP_SY0C;
    desc: Drop many files from Order intake (or straight into the shared folder). Picked up by the 60-second scan or Scan now.;
  elseif (Priority intake) than #blue
    [role_support: Pick files] <block_apply>
    label: Priority;
    props: APP_SA04;
    desc: Priority intake on the list takes several files. They skip the folder and appear in the list at once.;
  endif
  [role_system: Number and import] <block_system>
  label: Numbering;
  props: OA;
  desc: An order acceptance with an ORD- number is created in status Importing.;
  [role_system: AI extracts] <block_system>
  label: AI extraction;
  desc: Reads customer, item names, quantities, prices and dates. 1–3 min per document.;
  if (Extraction) is (Temporary failure) than #orange
    [role_system: Auto retry] <block_wait>
    label: Retrying;
    desc: Connection errors, congestion and malformed output are retried up to 3 times (orange Retrying).;
    [loop]
  elseif (Stopped) than #red
    [role_support: Retry or type in] <block_reject>
    label: Failed;
    props: APP_SA04;
    desc: The red Extraction failed badge gives cause and next step. Re-extract, or Switch to manual entry to make it a draft and type it in.;
  elseif (Success) than #blue
    [role_system: Match masters] <block_system>
    label: Matching;
    desc: Learned spellings first, then fuzzy matching; only a single hit is filled in. Status becomes Draft and the intake group is notified.;
  endif
elseif (By hand) than #purple
  [role_support: Create by hand] <block_apply>
  label: Manual;
  props: APP_SA04,OA;
  desc: Create by hand: enter customer and lines, then create the draft (status Draft).;
endif

label: Review;

  [role_support: Pick customer/item] <block_apply>

endif
