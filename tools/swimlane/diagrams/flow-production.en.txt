if (Where) is (In-house) than #blue
  if (Screen) is (Shared device) than #purple
    [role_mfg: Scan QR card] <block_apply>
    label: Log in;
    props: APP_KIOSK;
    desc: Works only on a shared device registered by an admin. Suspended or unassigned cards are refused.;
    if (PIN) is (Asked) than #orange
      [role_mfg: Enter PIN] <block_apply>
      label: PIN;
      props: APP_KIOSK;
      desc: Asked on first use of that device, after 48 hours without using that device, or 2 weeks after the last PIN. A new card sets its PIN by entering it twice.;
      if (Check) is (5 failures) than #red
        [role_system: Lock the card] <block_reject>
        label: 15-min lock;
        desc: Five wrong PINs in a row lock the card for 15 minutes. A forgotten PIN is reset by an admin.;
        [loop]
      elseif (Match) than
      endif
    elseif (Not asked) than
    endif
    [role_mfg: Open the step] <block_apply>
    label: Open step;
    props: APP_KIOSK;
    desc: "Steps" lists the steps planned for you as overdue / today / upcoming. "Work order scan" shows every step of a work order in step order from its QR (or number).;
  elseif (PC) than #purple
    [role_prod: Open step view] <block_apply>
    label: Step view;
    props: APP_PD02;
    desc: From the work order detail, "Open step execution view". View-only before approval and after completion.;
  endif
  if (Can start) is (No) than #orange
    [role_system: Block the start] <block_wait>
    label: Cannot start;
    desc: A step cannot start while the work order is not approved, a previous step / branch source / preceding work order is not completed, a required lot/slip code is empty, another user holds the session, or this work location may not run it.;
    [loop]
  elseif (Yes) than
  endif
  [role_mfg: Start the step] <block_apply>
  label: Start;
  props: APP_KIOSK,WO;
  desc: Check the received quantity (prefilled with the previous step's good quantity) and start. The step becomes In progress and is session-locked; the first start also puts the work order In progress.;
  if (Interrupt) is (Pause) than #blue
    [role_mfg: Pause] <block_wait>
    label: Pause;
    props: APP_KIOSK;
    desc: Time stops and the lock is released so someone else can take over. The step stays In progress.;
    [role_mfg: Resume] <block_apply>
    label: Resume;
    props: APP_KIOSK;
    desc: Time is added as a new work segment. Time on steps run at the same time is split by the number of concurrent steps.;
  elseif (Continue) than
  endif
elseif (Outsourced) than #blue
  [role_prod: Send out] <block_apply>
  label: Requested date;
  props: APP_PD02,MV;
  desc: Entering the requested date in the step's outsourcing dates moves the quantity to stock held by the subcontractor (stock movement "Issued to subcontractor"). The list is in Outsource order (PU04).;
  [role_supplier: Process and return] <block_neutral>
  label: Outsourced work;
  desc: The subcontractor does centerless grinding, coating and the like.;
  [role_prod: Log received date] <block_apply>
  label: Received date;
  props: APP_PD02,MV;
  desc: Entering the received date returns the held stock (stock movement "Returned from subcontractor"). Start and completion are recorded on the work order's step screen.;
endif
if (Step type) is (Inspection) than #blue
  [role_mfg: Enter inspection] <block_apply>
  label: Inspection;
  props: APP_KIOSK,IR;
  desc: Enter the inspected quantity and a measured value per item. Pass/fail is automatic against the tolerance; items that cannot be judged are set by hand. Sampling shows the required sample size. The step cannot complete until the sheet is saved.;
elseif (Insp. approval) than #blue
  [role_appr: Approve inspection] <block_approve>
  label: Insp. approval;
  props: APP_KIOSK,IR;
  desc: Approve after viewing the previous step's inspection sheet. Only passed records can be approved, and only by the sheet's approval group / approvers if set. No quantity is asked.;
elseif (Other) than
endif

label: Complete;

    [role_system: Cancel later steps] <block_end>

  endif

  [role_prod: Add branch series] <block_apply>
  label: Add branch;
  props: APP_PD02;
  desc: Sends the process-branch quantity through another series of steps. Choosing how it ends (merge into main line / to stock) is required.;
  if (Change approval) is (Pre-approval) than #purple
    [role_appr: Approve the change] <block_approve>
    label: Pre-approval;
    props: APP_PD02;
    desc: The change is held until approved and applied to the steps on final approval. Sending it back closes it unapplied.;
  elseif (Post-approval) than #purple
    [role_system: Apply at once] <block_system>
    label: Applied now;
    desc: The change is applied to the steps immediately.;
    [role_appr: Approve afterwards] <block_approve>
    label: Post-approval;
    props: APP_PD02;
    desc: If sent back, the steps are not reverted automatically. A red notice appears; check and press "Mark as checked".;
  elseif (No steps set) than
  endif
  [role_mfg: Run branch steps] <block_apply>
  label: Run branch;
  props: APP_KIOSK;
  desc: Branch steps are started and completed the same way, and cannot start until the branch source is completed.;
  if (Branch end) is (Merge) than #blue
    [role_system: Flow to merge step] <block_system>
    label: Merge;
    desc: The branch's good quantity becomes the received quantity of the chosen not-started main-line step.;

  endif

endif

  props: APP_PD02;
